# Vivo Debug RPK Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `vivo` mini-pack platform that packages the existing game as a debug `.rpk` while leaving every file under `games/` unchanged.

**Architecture:** Keep vivo support inside `mini-pack` and the root build wrapper. The game project remains read-only input; platform override, output directory selection, vivo manifest defaults, runtime bridge injection, and `mg-service build` execution are handled by the packager.

**Tech Stack:** TypeScript, Node.js 20, commander, esbuild, fs-extra, zod, vitest, `@vivo-minigame/cli-service` (`mg-service build`).

---

## Source Notes

The vivo CLI package documents `mg-service build` as the debug RPK build command. Use `@vivo-minigame/cli-service@1.27.13`, the npm version observed on 2026-05-09.

```js
import { spawnSync } from 'node:child_process';
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const repoRoot = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const { gamePath, platform } = parseArgs(process.argv.slice(2));

if (!gamePath) {
  console.error('Usage: pnpm build games/<game-project> [--platform douyin|vivo]');
  process.exit(1);
}

const supportedPlatforms = new Set(['douyin', 'vivo']);
if (!supportedPlatforms.has(platform)) {
  console.error(`Unsupported platform: ${platform}`);
  console.error('Supported platforms: douyin, vivo');
  process.exit(1);
}

const gameRoot = path.resolve(repoRoot, gamePath);
const gameName = path.basename(gameRoot);
const configFile = path.join(gameRoot, 'game.config.ts');
const outDir = path.posix.join('build', `${gameName}-${platform}`);

if (!fs.existsSync(configFile)) {
  console.error(`Config file not found: ${path.relative(repoRoot, configFile)}`);
  process.exit(1);
}

run('pnpm', ['--dir', 'mini-pack', 'build']);
run(process.execPath, [
  'mini-pack/dist/cli.js',
  'build',
  '--platform',
  platform,
  '--project-root',
  gamePath,
  '--out-dir',
  outDir,
]);

if (platform === 'douyin') {
  run(process.execPath, ['scripts/smoke-douyin.mjs', gamePath]);
}

function parseArgs(args) {
  let gamePathValue;
  let platformValue = 'douyin';

  for (let index = 0; index < args.length; index += 1) {
    const arg = args[index];
    if (arg === '--platform') {
      platformValue = args[index + 1] ?? '';
      index += 1;
      continue;
    }
    if (arg.startsWith('--platform=')) {
      platformValue = arg.slice('--platform='.length);
      continue;
    }
    if (!gamePathValue) {
      gamePathValue = arg;
      continue;
    }
    console.error(`Unexpected argument: ${arg}`);
    process.exit(1);
  }

  return {
    gamePath: gamePathValue,
    platform: platformValue,
  };
}

function run(command, args) {
  const result = spawnSync(command, args, {
    cwd: repoRoot,
    stdio: 'inherit',
  });

  if (result.status !== 0) {
    process.exit(result.status ?? 1);
  }
}
```

```ts
export function createVivoPackageName(projectRoot: string): string {
  const slug = path
    .basename(projectRoot)
    .toLowerCase()
    .replace(/[^a-z0-9]/g, '');
  return `com.minipack.${slug || 'game'}`;
}

export function renderVivoGameJs(bundleCode: string): string {
  return `/* Generated by mini-pack for vivo. */
(function () {
${bundleCode}

  function warn(message, data) {
    if (typeof console !== 'undefined' && typeof console.warn === 'function') {
      console.warn('[mini-pack:vivo]', message, data === undefined ? '' : data);
    }
  }

  function info(message, data) {
    if (typeof console !== 'undefined' && typeof console.info === 'function') {
      console.info('[mini-pack:vivo]', message, data === undefined ? '' : data);
    }
  }

  var root = typeof globalThis !== 'undefined' ? globalThis : this;
  var qg = root.qg || {};

  function createCanvas() {
    var canvas;
    if (qg && typeof qg.createCanvas === 'function') {
      canvas = qg.createCanvas();
      installTouchEventShim(canvas);
      return canvas;
    }
    if (root.canvas) {
      canvas = root.canvas;
      installTouchEventShim(canvas);
      return canvas;
    }
    if (typeof document !== 'undefined' && typeof document.createElement === 'function') {
      return document.createElement('canvas');
    }
    warn('Canvas API is unavailable; using a minimal fallback canvas.');
    return {
      width: 0,
      height: 0,
      getContext: function () {
        return null;
      }
    };
  }

  function installTouchEventShim(canvas) {
    if (!canvas || canvas.__miniPackVivoTouchShimInstalled) {
      return;
    }
    canvas.__miniPackVivoTouchShimInstalled = true;
    var nativeAddEventListener = typeof canvas.addEventListener === 'function' ? canvas.addEventListener.bind(canvas) : null;
    var nativeRemoveEventListener = typeof canvas.removeEventListener === 'function' ? canvas.removeEventListener.bind(canvas) : null;
    var eventMap = {
      pointerdown: ['onTouchStart', 'offTouchStart'],
      touchstart: ['onTouchStart', 'offTouchStart'],
      pointermove: ['onTouchMove', 'offTouchMove'],
      touchmove: ['onTouchMove', 'offTouchMove'],
      pointerup: ['onTouchEnd', 'offTouchEnd'],
      touchend: ['onTouchEnd', 'offTouchEnd'],
      pointercancel: ['onTouchCancel', 'offTouchCancel'],
      touchcancel: ['onTouchCancel', 'offTouchCancel']
    };
    var listeners = {};

    canvas.addEventListener = function (type, listener) {
      var mapping = eventMap[type];
      if (typeof listener !== 'function') {
        return;
      }
      if (!mapping) {
        if (nativeAddEventListener) {
          nativeAddEventListener(type, listener);
        }
        return;
      }
      var onName = mapping[0];
      if (typeof qg[onName] !== 'function') {
        if (nativeAddEventListener) {
          nativeAddEventListener(type, listener);
          return;
        }
        warn('Touch API is unavailable: ' + onName);
        return;
      }
      listeners[type] = listeners[type] || [];
      listeners[type].push(listener);
      qg[onName](listener);
    };

    canvas.removeEventListener = function (type, listener) {
      var mapping = eventMap[type];
      if (!mapping) {
        if (nativeRemoveEventListener && typeof listener === 'function') {
          nativeRemoveEventListener(type, listener);
        }
        return;
      }
      var list = listeners[type] || [];
      var index = list.indexOf(listener);
      if (index >= 0) {
        list.splice(index, 1);
      }
      var offName = mapping[1];
      if (typeof qg[offName] === 'function') {
        qg[offName](listener);
      }
    };
  }

  var memoryStorage = {};
  var storage = {
    getString: function (key) {
      try {
        if (typeof qg.getStorageSync === 'function') {
          var value = qg.getStorageSync(key);
          return typeof value === 'string' ? value : null;
        }
        if (root.localStorage) {
          return root.localStorage.getItem(key);
        }
      } catch (error) {
        warn('Storage get failed.', error);
      }
      return Object.prototype.hasOwnProperty.call(memoryStorage, key) ? memoryStorage[key] : null;
    },
    setString: function (key, value) {
      try {
        if (typeof qg.setStorageSync === 'function') {
          qg.setStorageSync(key, value);
          return;
        }
        if (root.localStorage) {
          root.localStorage.setItem(key, value);
          return;
        }
      } catch (error) {
        warn('Storage set failed.', error);
      }
      memoryStorage[key] = value;
    },
    remove: function (key) {
      try {
        if (typeof qg.removeStorageSync === 'function') {
          qg.removeStorageSync(key);
          return;
        }
        if (root.localStorage) {
          root.localStorage.removeItem(key);
          return;
        }
      } catch (error) {
        warn('Storage remove failed.', error);
      }
      delete memoryStorage[key];
    }
  };

  var muted = false;
  var sfxAudio = null;
  var musicAudio = null;

  function createAudioContext(label) {
    if (!qg || typeof qg.createInnerAudioContext !== 'function') {
      warn(label + ' audio API is unavailable.');
      return null;
    }
    var context = qg.createInnerAudioContext();
    if (context && typeof context.onError === 'function') {
      context.onError(function (error) {
        warn(label + ' audio playback failed.', error);
      });
    }
    return context;
  }

  function sfxPath(name) {
    return 'assets/audio/' + String(name).replace(/[^a-z0-9_-]/gi, '') + '.wav';
  }

  function musicPath(name) {
```

```ts
import { spawn } from 'node:child_process';

import { fileURLToPath } from 'node:url';

import { UserError } from '../../shared/errors.js';
import { createVivoPackageName } from './template.js';
import type { LoadedGameConfig } from '../../shared/types.js';

export interface VivoRpkBuildResult {
  rpkFiles: string[];
}

export async function buildVivoRpk(projectDir: string, config: LoadedGameConfig): Promise<VivoRpkBuildResult> {
  if (process.env.MINI_PACK_VIVO_FAKE_RPK === '1') {
    const fakeRpk = path.join(projectDir, 'dist/debug', `${createVivoPackageName(config.projectRoot)}.rpk`);
    await fs.mkdir(path.dirname(fakeRpk), { recursive: true });
    await fs.writeFile(fakeRpk, 'fake vivo rpk for tests\n');
    return { rpkFiles: [fakeRpk] };
  }

  const cliPath = await resolveVivoCliPath();
  await runCommand(cliPath, ['build'], projectDir);
  const rpkFiles = await findRpkFiles(projectDir);

  if (rpkFiles.length === 0) {
    throw new UserError(
      'Vivo CLI completed but no .rpk file was found.',
      `Check the generated project at ${projectDir} and run mg-service build manually for more details.`,
    );
  }

  return { rpkFiles };
}

async function resolveVivoCliPath(): Promise<string> {
  const packageRoot = await findMiniPackRoot(path.dirname(fileURLToPath(import.meta.url)));
  const executable = process.platform === 'win32' ? 'mg-service.cmd' : 'mg-service';
  const cliPath = path.join(packageRoot, 'node_modules', '.bin', executable);

  try {
    await fs.access(cliPath);
    return cliPath;
  } catch {
    throw new UserError(
      'Vivo CLI is not installed.',
      'Run pnpm --dir mini-pack install, then rerun pnpm build games/gonglian-fangxian --platform vivo.',
    );
  }
}

async function findMiniPackRoot(startDir: string): Promise<string> {
  let current = startDir;

  for (;;) {
    const packageJsonPath = path.join(current, 'package.json');
    try {
      const packageJson = JSON.parse(await fs.readFile(packageJsonPath, 'utf8')) as { name?: string };
      if (packageJson.name === 'mini-pack') {
        return current;
      }
    } catch {
      // Continue walking up.
    }

    const parent = path.dirname(current);
    if (parent === current) {
      throw new UserError('Unable to locate the mini-pack package root.');
    }
    current = parent;
  }
}

function runCommand(command: string, args: string[], cwd: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const child = spawn(command, args, {
      cwd,
      stdio: ['ignore', 'pipe', 'pipe'],
    });
    let stdout = '';
    let stderr = '';
    child.stdout.on('data', (chunk) => {
      stdout += chunk.toString();
    });
    child.stderr.on('data', (chunk) => {
      stderr += chunk.toString();
    });
    child.on('close', (exitCode) => {
      if (exitCode === 0) {
        resolve();
        return;
      }
      reject(
        new UserError(
          `Vivo CLI failed with exit code ${exitCode ?? 'unknown'}.`,
          `Command: ${command} ${args.join(' ')}\nProject: ${cwd}\nstdout:\n${stdout}\nstderr:\n${stderr}`,
        ),
      );
    });
    child.on('error', (error) => {
      reject(
        new UserError(
          `Failed to start Vivo CLI: ${error.message}`,
          `Command: ${command} ${args.join(' ')}\nProject: ${cwd}`,
        ),
      );
    });
  });
}

  const entries = await fs.readdir(dir, { recursive: true, withFileTypes: true });
  return entries
    .filter((entry) => entry.isFile() && entry.name.endsWith('.rpk'))
    .map((entry) => path.join(entry.parentPath, entry.name))
```
